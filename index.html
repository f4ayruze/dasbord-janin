MENARIK DATA RIWAYAT DARI FIREBASE --- // UBAH 1: Arahkan ke folder yang benar yaitu 'monitoring' var historyRef = database.ref('monitoring'); historyRef.on('value', (snapshot) => { var dataLabels = []; var dataGerakan = []; var dataSensor = []; var latestGerak = 0; var latestSensor = 0; // Siapkan baris pertama (Header) untuk file Excel/CSV csvData = [["Tanggal & Waktu", "Total Gerakan", "Nilai Sensor (BPM)"]]; snapshot.forEach((childSnapshot) => { var childData = childSnapshot.val(); var date = new Date(childData.timestamp); // Format jam untuk grafik (contoh: 10:45) var timeString = date.getHours().toString().padStart(2, '0') + ':' + date.getMinutes().toString().padStart(2, '0'); // Format lengkap untuk file Excel (contoh: 12/09/2024, 10:45:00) var fullDateString = date.toLocaleString('id-ID'); dataLabels.push(timeString); // UBAH 2: Samakan nama variabel dengan yang dikirim ESP32 (gerakan & bpm) dataGerakan.push(childData.gerakan); dataSensor.push(childData.bpm); // Memasukkan data ke dalam variabel array CSV csvData.push([fullDateString, childData.gerakan, childData.bpm]); latestGerak = childData.gerakan; latestSensor = childData.bpm; }); document.getElementById('gerakVal').innerText = latestGerak; document.getElementById('sensorVal').innerText = latestSensor; myChart.data.labels = dataLabels; myChart.data.datasets[0].data = dataGerakan; myChart.data.datasets[1].data = dataSensor; myChart.update(); }); // --- 4. FUNGSI UNTUK MENDOWNLOAD CSV --- function downloadCSV() { // Peringatan jika data dari Firebase belum masuk sama sekali if (csvData.length <= 1) { alert("Data masih kosong atau sedang dimuat. Harap tunggu hingga grafik muncul."); return; } var csvContent = ""; // Menggabungkan array menjadi format text yang dipisahkan koma csvData.forEach(function(rowArray) { var row = rowArray.join(","); csvContent += row + "\r\n"; }); // Proses konversi dan pembuatan file unduhan var blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' }); var link = document.createElement("a"); var url = URL.createObjectURL(blob); link.setAttribute("href", url); link.setAttribute("download", "Data_Monitoring_Janin_Device20.csv"); link.style.visibility = 'hidden'; document.body.appendChild(link); link.click(); document.body.removeChild(link); }
